    setState(() {
      _isLoading = true;
      _error = '';
    });

    try {
      final probeResult = await Process.run(
        'ffprobe',
        [
          '-v', 'error',
          '-show_entries', 'stream=index,codec_type',
          '-of', 'csv=p=0',
          widget.videoPath
        ],
        runInShell: true,
      );

      final output = probeResult.stdout.toString();
      final lines = output.trim().split('\n');
      
      final List<int> audioIndexes = [];
      for (final line in lines) {
        final parts = line.split(',');
        if (parts.length >= 2 && parts[1].trim() == 'audio') {
          final index = int.tryParse(parts[0].trim());
          if (index != null) {
            audioIndexes.add(index);
          }
        }
      }

      if (audioIndexes.isEmpty) {
        setState(() {
          _error = 'Аудиодорожки не найдены';
          _isLoading = false;
        });
        return;
      }

      final List<AudioTrack> tracks = [];
      for (final idx in audioIndexes) {
        double meanVolume = -100;
        
        try {
          final volumeResult = await Process.run(
            'ffmpeg',
            [
              '-i', widget.videoPath,
              '-map', '0:a:$idx',
              '-af', 'volumedetect',
              '-f', 'null',
              'NUL'
            ],
            runInShell: true,
          );

          final stderr = volumeResult.stderr.toString();
          final meanMatch = RegExp(r'mean_volume:\s*(-?\d+(?:\.\d+)?)').firstMatch(stderr);
          if (meanMatch != null) {
            meanVolume = double.parse(meanMatch.group(1)!);
          }
        } catch (e) {
          // Игнорируем
        }

        tracks.add(AudioTrack(
          index: idx,
          name: 'Дорожка ${idx + 1}',
          isEnabled: true,
          volumePercent: 100,
          meanVolume: meanVolume,
        ));
      }

        _tracks = tracks;
        _isLoading = false;

import 'dart:math';

class AudioTrack {
  final int index;
  final String name;
  bool isEnabled;
  double volumePercent; // 0-200
  final double originalVolume; // dB из volumedetect

  AudioTrack({
    required this.index,
    required this.name,
    required this.isEnabled,
    required this.volumePercent,
    required this.originalVolume,
  });

  double get finalVolumeDb {
    if (volumePercent == 100) return originalVolume;
    return originalVolume + 20 * log(volumePercent / 100) / ln10;
  }

  String get volumeFilter {
    if (!isEnabled) return 'volume=0';
    if (volumePercent == 100) return '';
    final factor = volumePercent / 100;
    return 'volume=${factor.toStringAsFixed(2)}';
  }
}

import 'dart:io';
import '../models/audio_track.dart';

class FFmpegService {
  static Future<List<AudioTrack>> analyzeAudio(String videoPath) async {
    final probeResult = await Process.run(
      'ffprobe',
      [
        '-v', 'error',
        '-show_entries', 'stream=index,codec_type',
        '-of', 'csv=p=0',
        videoPath
      ],
      runInShell: true,
    );

    final lines = probeResult.stdout.toString().trim().split('\n');
    final List<int> audioIndexes = [];
    
    for (final line in lines) {
      final parts = line.split(',');
      if (parts.length >= 2 && parts[1].trim() == 'audio') {
        final index = int.tryParse(parts[0].trim());
        if (index != null) audioIndexes.add(index);
      }
    }

    final List<AudioTrack> tracks = [];
    for (final idx in audioIndexes) {
      double originalVolume = -100;
      try {
        final volumeResult = await Process.run(
          'ffmpeg',
          [
            '-i', videoPath,
            '-map', '0:a:$idx',
            '-af', 'volumedetect',
            '-f', 'null',
            'NUL'
          ],
          runInShell: true,
        );
        final match = RegExp(r'mean_volume:\s*(-?\d+(?:\.\d+)?)')
            .firstMatch(volumeResult.stderr.toString());
        if (match != null) {
          originalVolume = double.parse(match.group(1)!);
        }
      } catch (e) {}
      
      tracks.add(AudioTrack(
        index: idx,
        name: 'Дорожка ${idx + 1}',
        isEnabled: true,
        volumePercent: 100,
        originalVolume: originalVolume,
      ));
    }
    return tracks;
  }

  static Future<bool> exportVideo({
    required String inputPath,
    required String outputPath,
    required List<AudioTrack> audioTracks,
    double? trimSeconds,
    int? trimMode,
  }) async {
    final List<String> args = ['-i', inputPath];
    
    // Собираем аудио фильтры
    final List<String> audioFilters = [];
    final List<String> audioInputs = [];
    
    for (int i = 0; i < audioTracks.length; i++) {
      final track = audioTracks[i];
      final filter = track.volumeFilter;
      if (filter.isNotEmpty) {
        audioFilters.add('[0:a:${track.index}]$filter[a$i]');
        audioInputs.add('[a$i]');
      } else if (track.isEnabled) {
        audioInputs.add('[0:a:${track.index}]');
      } else {
        audioFilters.add('[0:a:${track.index}]volume=0[a$i]');
        audioInputs.add('[a$i]');
      }
    }
    
    // Добавляем обрезку если нужно
    if (trimSeconds != null && trimSeconds > 0 && trimMode == 0) {
      args.addAll(['-t', trimSeconds.toStringAsFixed(2)]);
    }
    
    // Добавляем аудио фильтры
    if (audioInputs.isNotEmpty) {
      final filterComplex = '${audioFilters.join('; ')}; ${audioInputs.join('')}amix=inputs=${audioInputs.length}:duration=longest[aout]';
      args.addAll(['-filter_complex', filterComplex, '-map', '[aout]']);
    } else {
      args.add('-an');
    }
    
    args.addAll(['-c:v', 'copy', '-y', outputPath]);
    
    final result = await Process.run('ffmpeg', args, runInShell: true);
    return result.exitCode == 0;
  }
}

    final dir = await getDownloadsDirectory();
    final outPath = '${dir!.path}\\${DateTime.now().millisecondsSinceEpoch}_exported.mp4';
    final success = await FFmpegService.exportVideo(
      inputPath: _videoPath!,
      outputPath: outPath,
      audioTracks: _audioTracks,
      trimSeconds: _trimSeconds > 0 ? _trimSeconds : null,
      trimMode: _trimMode,
    );
    setState(() => _isLoading = false);
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(success ? '✅ Экспорт завершён!\n$outPath' : '❌ Ошибка экспорта')),
    );
  }
